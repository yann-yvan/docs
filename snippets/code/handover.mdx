```bash cURL
# Take the conversation over: the AI agent stops answering
curl -X PUT https://shortext.ny-corp.io/api/v2.0/conversations/$CONVERSATION_ID/handover \
  -H "Authorization: Bearer $SHORTEXT_API_KEY" -H "Accept: application/json"

# Hand it back: the AI agent answers again
curl -X DELETE https://shortext.ny-corp.io/api/v2.0/conversations/$CONVERSATION_ID/handover \
  -H "Authorization: Bearer $SHORTEXT_API_KEY" -H "Accept: application/json"
```

```php PHP
<?php
$client  = new GuzzleHttp\Client(['base_uri' => 'https://shortext.ny-corp.io/api/v2.0/']);
$headers = ['Authorization' => 'Bearer ' . getenv('SHORTEXT_API_KEY'), 'Accept' => 'application/json'];

$client->put("conversations/{$conversationId}/handover", ['headers' => $headers]);    // take over
$client->delete("conversations/{$conversationId}/handover", ['headers' => $headers]); // hand back
```
