```php PHP
<?php
$response = (new GuzzleHttp\Client())->get('https://shortext.ny-corp.io/api/v2.0/me', [
    'headers' => [
        'Authorization' => 'Bearer ' . getenv('SHORTEXT_API_KEY'),
        'Accept'        => 'application/json',
    ],
]);

$result = json_decode((string) $response->getBody(), true);
if (!$result['success']) {
    throw new RuntimeException("Shortext error {$result['code']}: {$result['message']}");
}
echo $result['body']['company']['name'];
```

```javascript Node.js
const response = await fetch("https://shortext.ny-corp.io/api/v2.0/me", {
  headers: {
    Authorization: `Bearer ${process.env.SHORTEXT_API_KEY}`,
    Accept: "application/json",
  },

}
```

```python Python
import os
import requests

response = requests.get(
    "https://shortext.ny-corp.io/api/v2.0/me",
    headers={
        "Authorization": f"Bearer {os.environ['SHORTEXT_API_KEY']}",
        "Accept": "application/json",
    },
)

result = response.json()
if not result["success"]:
    raise RuntimeError(f"Shortext error {result['code']}: {result['message']}")
print(result["body"]["company"]["name"])
```
