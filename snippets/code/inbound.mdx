```php PHP
<?php
$response = (new GuzzleHttp\Client())->post('https://shortext.ny-corp.io/api/v1.0/messages/inbound', [
    'headers' => ['Authorization' => 'Bearer ' . getenv('SHORTEXT_API_KEY'), 'Accept' => 'application/json'],
    'json'    => [
        'channel'    => 'shortext',
        'to'         => 'ste_xxx',
        'reply_mode' => 'sync',
        'contact'    => ['identifier' => 'user-42', 'profile' => ['firstname' => 'Awa']],
        'message'    => ['id' => 'msg_9f8e7d', 'type' => 'text', 'content' => ['text' => 'Where is my order?']],
    ],
]);
$body = json_decode((string) $response->getBody(), true)['body'];

if ($body['status'] === 'completed') {
    foreach ($body['replies'] as $reply) {
        echo $reply['message']['content']['text'], PHP_EOL;
    }
}
```
